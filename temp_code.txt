@Composable
fun UpcomingMedicationsSection(
    pendingMedicines: List<com.critetiontech.ctvitalio.model.PendingMedicine>,
    markingMedicines: Set<Int>,
    onMarkIntake: (com.critetiontech.ctvitalio.model.PendingMedicine) -> Unit
) {
    val colors = LocalMyColorScheme.current
    if (pendingMedicines.isEmpty()) return

    Column(modifier = Modifier.fillMaxWidth()) {
        Text("Upcoming Medications", style = AppTextStyles.style18BCB())
        Spacer(Modifier.height(12.dp))

        pendingMedicines.forEach { medicine ->
            val isMarking = markingMedicines.contains(medicine.medicineId)
            Card(
                shape = RoundedCornerShape(12.dp),
                colors = CardDefaults.cardColors(containerColor = colors.dashboardContainerColor),
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(bottom = 8.dp)
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth().padding(16.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Box(
                        modifier = Modifier.size(48.dp).clip(RoundedCornerShape(12.dp)).background(Color(0xFFEFF6FF)),
                        contentAlignment = Alignment.Center
                    ) {
                        Image(painter = painterResource(id = R.drawable.medicine_reminder), contentDescription = null, modifier = Modifier.size(24.dp))
                    }
                    Spacer(Modifier.width(16.dp))
                    Column(modifier = Modifier.weight(1f)) {
                        Text("1 unit", style = AppTextStyles.style12BCN().copy(color = Color(0xFF64748B)))
                        Spacer(Modifier.height(4.dp))
                        Text(medicine.medicineName, style = AppTextStyles.style14BCB().copy(color = Color(0xFF0F172A), fontWeight = FontWeight.Bold))
                        Spacer(Modifier.height(4.dp))
                        Text(if (medicine.instructions.isNotEmpty()) medicine.instructions else medicine.frequency, style = AppTextStyles.style12GCN().copy(color = Color(0xFF64748B)))
                    }
                    Spacer(Modifier.width(16.dp))
                    Column(
                        horizontalAlignment = Alignment.End,
                        verticalArrangement = Arrangement.Center,
                        modifier = Modifier.height(64.dp)
                    ) {
                        if (isMarking) {
                            CircularProgressIndicator(modifier = Modifier.size(24.dp), color = Color(0xFF4F46E5), strokeWidth = 2.dp)
                        } else {
                            Box(
                                modifier = Modifier.size(24.dp).border(1.5.dp, Color(0xFFCBD5E1), RoundedCornerShape(6.dp)).clickable { onMarkIntake(medicine) }
                            )
                        }
                    }
                }
            }
        }
    }
}
